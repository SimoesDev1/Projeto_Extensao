def _registrar_memoria_ocr(etapa):
    """Registra o consumo de memoria em uma etapa do OCR."""
    from pathlib import Path

    memoria_processo = "N/D"
    uso_container = "N/D"
    limite_container = "N/D"

    try:
        status = Path("/proc/self/status")
        if status.exists():
            for linha in status.read_text().splitlines():
                if linha.startswith("VmRSS:"):
                    memoria_processo = round(
                        int(linha.split()[1]) / 1024, 2
                    )
                    break
    except Exception:
        pass

    try:
        uso = Path("/sys/fs/cgroup/memory.current")
        limite = Path("/sys/fs/cgroup/memory.max")

        if uso.exists():
            uso_container = round(
                int(uso.read_text().strip()) / (1024 ** 2), 2
            )

        if limite.exists():
            valor = limite.read_text().strip()
            if valor != "max":
                limite_container = round(
                    int(valor) / (1024 ** 2), 2
                )
        else:
            uso_v1 = Path(
                "/sys/fs/cgroup/memory/memory.usage_in_bytes"
            )
            limite_v1 = Path(
                "/sys/fs/cgroup/memory/memory.limit_in_bytes"
            )

            if uso_v1.exists():
                uso_container = round(
                    int(uso_v1.read_text().strip()) / (1024 ** 2), 2
                )

            if limite_v1.exists():
                limite_container = round(
                    int(limite_v1.read_text().strip()) / (1024 ** 2), 2
                )

    except Exception:
        pass

    logger.info(
        "OCR | Memoria | Etapa: %s | "
        "RAM processo: %s MB | "
        "Uso container: %s MB | "
        "Limite container: %s MB",
        etapa,
        memoria_processo,
        uso_container,
        limite_container
    )

    try:

    except Exception:
        pass

    try:

        uso = Path("/sys/fs/cgroup/memory.current")

        if limite.exists():

class PlanilhaProdutoService:
    # VariaÃ§Ãµes comuns de nomes de colunas
    COLUNAS_PRODUTO_ALIASES = [
        'produto', 'nome', 'descricao', 'descriÃ§Ã£o', 'item', 'titulo',
        'mercadoria', 'artigo', 'servico', 'serviÃ§o', 'especificacao',
        'especificaÃ§Ã£o', 'nome_produto', 'nome do produto', 'produtos'
    ]

    COLUNAS_PRECO_ALIASES = [
        'preco', 'preÃ§o', 'valor', 'preco_venda', 'preÃ§o_venda',
        'preco_unitario', 'preÃ§o_unitÃ¡rio', 'preco_atual', 'preÃ§o_atual',
        'valor_unitario', 'valor_unitÃ¡rio', 'valor_venda', 'preco_tabela',
        'custo', 'preco_custo', 'preco_original', 'preÃ§o_original',
        'precos', 'preÃ§os', 'valores'
    ]

    COLUNAS_CODIGO_ALIASES = [
        'codigo', 'cÃ³digo', 'id', 'cod', 'sku', 'ref', 'referencia', 'referÃªncia'
    ]

    COLUNAS_CATEGORIA_ALIASES = [
        'categoria', 'tipo', 'grupo', 'departamento', 'secao', 'seÃ§Ã£o', 'linha'

    @classmethod
    def normalizar_texto(cls, texto) -> str:
        """Remove acentos e caracteres especiais para comparaÃ§Ã£o flexÃ­vel."""
        if isinstance(texto, (pd.Series, list, tuple)):
            texto = texto[0] if len(texto) > 0 else ''
        if not isinstance(texto, str):
            texto = str(texto) if (texto is not None and pd.notna(texto)) else ''
        t = texto.strip().lower()
        t = re.sub(r'[Ã¡Ã Ã¢Ã£Ã¤]', 'a', t)
        t = re.sub(r'[Ã©Ã¨ÃªÃ«]', 'e', t)
        t = re.sub(r'[Ã­Ã¬Ã®Ã¯]', 'i', t)
        t = re.sub(r'[Ã³Ã²Ã´ÃµÃ¶]', 'o', t)
        t = re.sub(r'[ÃºÃ¹Ã»Ã¼]', 'u', t)
        t = re.sub(r'[Ã§]', 'c', t)
        t = re.sub(r'[^a-z0-9]', '', t)
        return t

    @classmethod
    def deduplicar_colunas(cls, colunas) -> list[str]:
        """Garante que todos os nomes de colunas sejam strings Ãºnicas e limpas."""
        contagem = {}
        novas = []
        for i, c in enumerate(colunas):
            if c is not None and not (isinstance(c, float) and math.isnan(c)) and str(c).strip() != '':
                nome = str(c).strip()
            else:
                nome = f"Coluna_{i+1}"

            if nome in contagem:
                contagem[nome] += 1
                novas.append(f"{nome}_{contagem[nome]}")
            else:
                contagem[nome] = 0
                novas.append(nome)
        return novas

    @classmethod
    def _obter_escalar(cls, val, default: str = "") -> str:
        """Garante retorno de string escalar segura, desempacotando Series caso ocorra."""
        if isinstance(val, (pd.Series, list, tuple)):
            for v in val:
                if v is not None and not pd.isna(v) and str(v).strip() != '':
                    return str(v).strip()
            return default
        if val is None or pd.isna(val):
            return default
        s = str(val).strip()
        return s if s != '' else default

    @classmethod
    def identificar_colunas(cls, colunas: list[str]) -> dict:
        """
        Identifica automaticamente quais colunas correspondem a Produto, PreÃ§o, CÃ³digo e Categoria.
        """
        mapeamento = {
            'produto': None,
            'preco': None,
            'codigo': None,
            'categoria': None
        }

        colunas_norm = {col: cls.normalizar_texto(col) for col in colunas}

        # 1. Procura Coluna de PreÃ§o (prioritÃ¡rio)
        for col_orig, col_n in colunas_norm.items():
            for alias in cls.COLUNAS_PRECO_ALIASES:
                if cls.normalizar_texto(alias) == col_n or cls.normalizar_texto(alias) in col_n:
                    mapeamento['preco'] = col_orig
                    break
            if mapeamento['preco']:
                break

        # 2. Procura Coluna de Produto
        for col_orig, col_n in colunas_norm.items():
            if col_orig == mapeamento['preco']:
                continue
            for alias in cls.COLUNAS_PRODUTO_ALIASES:
                if cls.normalizar_texto(alias) == col_n or cls.normalizar_texto(alias) in col_n:
                    mapeamento['produto'] = col_orig
                    break
            if mapeamento['produto']:
                break

        # 3. Procura CÃ³digo (opcional)
        for col_orig, col_n in colunas_norm.items():
            if col_orig in (mapeamento['preco'], mapeamento['produto']):
                continue
            for alias in cls.COLUNAS_CODIGO_ALIASES:
                if cls.normalizar_texto(alias) == col_n:
                    mapeamento['codigo'] = col_orig
                    break
            if mapeamento['codigo']:
                break

        # 4. Procura Categoria (opcional)
        for col_orig, col_n in colunas_norm.items():
            if col_orig in (mapeamento['preco'], mapeamento['produto'], mapeamento['codigo']):
                continue
            for alias in cls.COLUNAS_CATEGORIA_ALIASES:
                if cls.normalizar_texto(alias) == col_n:
                    mapeamento['categoria'] = col_orig
                    break
            if mapeamento['categoria']:
                break

        # Fallback se nÃ£o encontrou coluna de produto, usa a primeira coluna nÃ£o de preÃ§o
        if not mapeamento['produto'] and colunas:
            for c in colunas:
                if c != mapeamento['preco']:
                    mapeamento['produto'] = c
                    break

        return mapeamento

    @classmethod
    def converter_valor_monetario(cls, valor) -> float | None:
        """
        Converte string, nÃºmero ou item de Series para float.
        Ex: 'R$ 1.250,50' -> 1250.50, '15,90' -> 15.90, 15.9 -> 15.9
        """
        if isinstance(valor, (pd.Series, list, tuple)):
            for v in valor:
                res = cls.converter_valor_monetario(v)
                if res is not None:
                    return res
            return None

        if valor is None or pd.isna(valor):
            return None

        if isinstance(valor, (int, float)):
            return float(valor)

        texto = str(valor).strip()
        if not texto:
            return None

        texto = texto.replace('R$', '').replace('r$', '').replace('\xa0', '').strip()

        if ',' in texto:
            texto = texto.replace('.', '')
            texto = texto.replace(',', '.')
        else:
            if re.search(r'\d+,\d{3}', texto):
                texto = texto.replace(',', '')

        try:
            val = float(re.sub(r'[^\d.-]', '', texto))
            return val
        except (ValueError, TypeError):
            return None

    @classmethod
    def aplicar_regra_arredondamento(cls, valor: float, regra: str = 'padrao') -> float:
        """
        Aplica regras de precificaÃ§Ã£o mercadolÃ³gica:
        - 'padrao': 2 casas decimais normais (ex: R$ 15,48)
        - 'termina_99': arredonda para centavos .99 (ex: R$ 15,99)
        - 'termina_90': arredonda para centavos .90 (ex: R$ 15,90)
        - 'inteiro': nÃºmero inteiro mais prÃ³ximo (ex: R$ 15,00)
        """
        if regra == 'termina_99':
            if valor <= 1.0:
                return round(valor, 2)
            base = math.floor(valor)
            if (valor - base) > 0.99:
                return float(base + 1.99)
            return float(base + 0.99)

        elif regra == 'termina_90':
            if valor <= 1.0:
                return round(valor, 2)
            base = math.floor(valor)
            if (valor - base) > 0.90:
                return float(base + 1.90)
            return float(base + 0.90)

        elif regra == 'inteiro':
            return float(round(valor))

        return round(valor, 2)

    @classmethod
    def _validar_mes_vigencia(cls, mes_vigencia: str | None) -> tuple[int, int]:
        """
        Converte 'AAAA-MM' (formato do <input type="month">) em (ano, mes).
        Sem valor, usa o mÃªs atual.
        """
        if not mes_vigencia:
            agora = datetime.now()
            return agora.year, agora.month
        try:
            ano_str, mes_str = str(mes_vigencia).strip().split('-')
            ano, mes = int(ano_str), int(mes_str)
            if 1 <= mes <= 12 and 2000 <= ano <= 2100:
                return ano, mes
        except ValueError:
            pass
        raise ValueError("MÃªs de vigÃªncia invÃ¡lido. Use o formato AAAA-MM (ex.: 2026-11).")

    @classmethod
    def _detectar_e_ajustar_cabecalho(cls, df_raw: pd.DataFrame) -> pd.DataFrame:
        """
        Identifica em qual linha estÃ¡ o cabeÃ§alho real caso haja banners, tÃ­tulos ou linhas vazias antes da tabela.
        Garante que todos os nomes de colunas sejam strings deduplicadas.
        """
        if df_raw.empty:
            return df_raw

        max_linhas = min(12, len(df_raw))
        melhor_linha = -1
        max_matches = 0

        aliases_todos = cls.COLUNAS_PRECO_ALIASES + cls.COLUNAS_PRODUTO_ALIASES + cls.COLUNAS_CODIGO_ALIASES

        for idx in range(max_linhas):
            row_vals = [cls.normalizar_texto(v) for v in df_raw.iloc[idx]]
            matches = sum(1 for val in row_vals if any(cls.normalizar_texto(a) == val or cls.normalizar_texto(a) in val for a in aliases_todos))
            if matches > max_matches:
                max_matches = matches
                melhor_linha = idx

        if melhor_linha >= 0 and max_matches > 0:
            raw_cols = df_raw.iloc[melhor_linha].tolist()
            novas_colunas = cls.deduplicar_colunas(raw_cols)
            df_ajustado = df_raw.iloc[melhor_linha + 1:].reset_index(drop=True)
            df_ajustado.columns = novas_colunas

            # Se alguma linha de dados for repetiÃ§Ã£o do cabeÃ§alho, descarta
            filtro = []
            for _, r in df_ajustado.iterrows():
                r_vals = [str(v).strip() for v in r.tolist()]
                if r_vals == [str(c).strip() for c in raw_cols]:
                    filtro.append(False)
                else:
                    filtro.append(True)
            df_ajustado = df_ajustado[filtro].reset_index(drop=True)
            return df_ajustado

        primeira_linha = df_raw.iloc[0].tolist()
        tem_preco_linha_0 = any(cls.converter_valor_monetario(v) is not None for v in primeira_linha)
        if tem_preco_linha_0:
            if len(df_raw.columns) == 2:
                novas_colunas = ['Produto', 'PreÃ§o']
            elif len(df_raw.columns) == 3:
                novas_colunas = ['CÃ³digo', 'Produto', 'PreÃ§o']
            else:
                novas_colunas = [f'Coluna_{i+1}' for i in range(len(df_raw.columns))]
            df_raw.columns = cls.deduplicar_colunas(novas_colunas)
            return df_raw
        else:
            df_raw.columns = cls.deduplicar_colunas(primeira_linha)
            return df_raw.iloc[1:].reset_index(drop=True)

    @classmethod
    def _preparar_imagem_para_ocr(cls, conteudo: bytes):
        """Abre a imagem, corrige a rotaÃ§Ã£o (fotos de celular) e amplia imagens pequenas.
        Prints pequenos (ex.: 215 px de largura) perdem espaÃ§os e vÃ­rgulas no OCR sem isso."""
        img = ImageOps.exif_transpose(Image.open(io.BytesIO(conteudo))).convert('RGB')
        max_lado = int(os.getenv("OCR_MAX_LADO", "1600"))
        if max(img.size) > max_lado:
            img.thumbnail((max_lado, max_lado), Image.LANCZOS)
        fator = min(3, max(1, round(max_lado / max(img.size))))
        if fator > 1:
            img = img.resize((img.width * fator, img.height * fator), Image.LANCZOS)
        return np.array(img)

    @classmethod
    def ler_arquivo_imagem(cls, stream) -> pd.DataFrame:
        """
        Extrai produtos e preÃ§os de fotos e prints via OCR (RapidOCR).
        Funciona para tabelas (produto e preÃ§o em caixas separadas) e para listas
        em texto corrido ("Produto - 10,95"), onde o OCR devolve a linha inteira numa caixa sÃ³.
        """
        conteudo = stream.read()
        stream.seek(0)

        # Cache de OCR por SHA-256 do arquivo (mÃ¡x. 8 entradas)
        hash_arquivo = hashlib.sha256(conteudo).hexdigest()
        cached_df = _obter_ocr_cache(f"df_{hash_arquivo}")
        if cached_df is not None:
            logger.info("OCR | Reutilizando resultado do cache SHA-256: %s...", hash_arquivo[:12])
            return cached_df.copy()

        if not _OCR_PIPELINE_LOCK.acquire(blocking=False):
            raise OCROcupadoError(
                "O OCR esta processando outro documento. Aguarde alguns instantes e tente novamente."
            )

        try:
            try:
                inicio_preparacao = time.perf_counter()

                imagem = cls._preparar_imagem_para_ocr(conteudo)

                tempo_preparacao = time.perf_counter() - inicio_preparacao

                logger.info(
                    "OCR | Preparacao da imagem: %.2fs | Dimensoes: %s x %s | "
                    "Memoria: %.2f MB | Arquivo original: %.2f KB",
                    tempo_preparacao,
                    imagem.shape[1],
                    imagem.shape[0],
                    imagem.nbytes / (1024 * 1024),
                    len(conteudo) / 1024
                )

            except Exception:
                raise ValueError(
                    "Nao foi possivel abrir a imagem enviada. "
                    "Envie um arquivo PNG, JPG ou WEBP valido."
                )

            try:
                res, _ = _executar_ocr(imagem)
            finally:
                del imagem

        finally:
            _OCR_PIPELINE_LOCK.release()

        if not res:
            raise ValueError("NÃ£o foi possÃ­vel detectar texto ou nÃºmeros na imagem enviada. Certifique-se de que a imagem esteja nÃ­tida.")

        caixas = []
        for box, texto, *_ in res:
            texto = str(texto).strip()
            if not texto:
                continue
            ys = [p[1] for p in box]
            xs = [p[0] for p in box]
            caixas.append({'texto': texto, 'y': (min(ys) + max(ys)) / 2.0,
                           'x': min(xs), 'h': max(ys) - min(ys)})
        if not caixas:
            raise ValueError("NÃ£o foi possÃ­vel detectar texto ou nÃºmeros na imagem enviada. Certifique-se de que a imagem esteja nÃ­tida.")

        # TolerÃ¢ncia de "mesma linha" proporcional Ã  altura do texto (nÃ£o mais 18 px fixos)
        altura_mediana = sorted(c['h'] for c in caixas)[len(caixas) // 2]
        tolerancia = max(4.0, 0.6 * altura_mediana)

        caixas.sort(key=lambda c: c['y'])
        linhas = []
        for c in caixas:
            if linhas and abs(linhas[-1]['y'] - c['y']) <= tolerancia:
                linhas[-1]['itens'].append(c)
                linhas[-1]['y'] += (c['y'] - linhas[-1]['y']) / len(linhas[-1]['itens'])
            else:
                linhas.append({'y': c['y'], 'itens': [c]})

        cabecalhos = ('produto', 'produtos', 'item', 'descricao', 'nome')
        produtos_extraidos = []
        sem_preco = []
        for linha in linhas:
            texto_linha = ' '.join(i['texto'] for i in sorted(linha['itens'], key=lambda i: i['x'])).strip()

            m = _LINHA_PRODUTO_PRECO.match(texto_linha)
            if m:
                nome, preco = m['nome'], m['preco']
            else:
                achados = list(_PRECO_QUALQUER.finditer(texto_linha))
                if not achados:
                    sem_preco.append(texto_linha)
                    continue
                ultimo = achados[-1]
                nome, preco = texto_linha[:ultimo.start()], ultimo.group(1)

            nome = nome.strip(' â€¢Â·.:+*-â€“â€”|;\t')
            if not re.search(r'[A-Za-zÃ€-Ã¿]', nome) or cls.normalizar_texto(nome) in cabecalhos:
                continue
            produtos_extraidos.append([nome, preco])

        logger.info("OCR de imagem: %d produtos; linhas sem preÃ§o ignoradas: %s",
                    len(produtos_extraidos), sem_preco)

        if not produtos_extraidos:
            raise ValueError("NÃ£o foi possÃ­vel identificar pares de produto e preÃ§o na imagem/print. Verifique se a imagem contÃ©m preÃ§os visÃ­veis com centavos.")

        df_extraido = pd.DataFrame(produtos_extraidos, columns=['Produto', 'PreÃ§o'])
        _salvar_ocr_cache(f"df_{hash_arquivo}", df_extraido)
        return df_extraido

    @classmethod
    def ler_arquivo_pdf(cls, stream) -> pd.DataFrame:
        """
        Extrai tabelas de arquivos PDF utilizando pdfplumber.
        """
        linhas_brutas = []

        with pdfplumber.open(stream) as pdf:
            for page in pdf.pages:
                raw_tables = page.extract_tables() or []
                valid_tables = [t for t in raw_tables if any(len(r) >= 2 for r in t)]

                if valid_tables:
                    for tabela in valid_tables:
                        for row in tabela:
                            if not row:
                                continue
                            row_clean = [str(cell).strip() if cell is not None else '' for cell in row]
                            if any(row_clean):
                                linhas_brutas.append(row_clean)
                else:
                    texto = page.extract_text() or ''
                    for line in texto.split('\n'):
                        line = line.strip()
                        if not line:
                            continue
                        m_preco = re.search(r'(\d{1,3}(?:\.\d{3})*,\d{2}|\d+\.\d{2})\s*$', line)
                        if m_preco:
                            preco_str = m_preco.group(1)
                            resto = line[:m_preco.start()].strip(' -:;\t|')
                            resto = re.sub(r'R\$\s*$', '', resto, flags=re.IGNORECASE).strip(' -:;\t|')
                            m_cod = re.match(r'^([A-Z0-9_-]{3,10})\s+[-â€“]?\s*(.+)$', resto, re.IGNORECASE)
                            if m_cod:
                                cod = m_cod.group(1)
                                prod = m_cod.group(2)
                                linhas_brutas.append([cod, prod, preco_str])
                            else:
                                linhas_brutas.append([resto, preco_str])

        if not linhas_brutas:
            raise ValueError("NÃ£o foi possÃ­vel identificar nenhuma tabela ou lista de produtos legÃ­vel no arquivo PDF enviado.")

        max_cols = max(len(r) for r in linhas_brutas)
        linhas_padronizadas = [r + [''] * (max_cols - len(r)) for r in linhas_brutas]

        df_raw = pd.DataFrame(linhas_padronizadas)
        return cls._detectar_e_ajustar_cabecalho(df_raw)

    @classmethod
    def ler_arquivo(cls, stream, nome_arquivo: str) -> pd.DataFrame:
        """
        LÃª tabela nos formatos .png, .jpg, .jpeg, .webp (imagem/print), .pdf, .xlsx, .xls ou .csv.
        """
        ext = nome_arquivo.rsplit('.', 1)[-1].lower() if '.' in nome_arquivo else ''

        if ext in ('png', 'jpg', 'jpeg', 'webp'):
            return cls.ler_arquivo_imagem(stream)
        elif ext == 'pdf':
            return cls.ler_arquivo_pdf(stream)
        elif ext in ('xlsx', 'xls'):
            df_raw = pd.read_excel(stream, header=None)
            return cls._detectar_e_ajustar_cabecalho(df_raw)
        elif ext == 'csv':
            conteudo = stream.read()
            stream.seek(0)
            for encoding in ['utf-8-sig', 'utf-8', 'latin-1', 'cp1252']:
                for sep in [';', ',', '\t']:
                    try:
                        df_raw = pd.read_csv(io.BytesIO(conteudo), sep=sep, encoding=encoding, header=None)
                        if len(df_raw.columns) > 1 or len(df_raw) > 0:
                            return cls._detectar_e_ajustar_cabecalho(df_raw)
                    except Exception:
                        continue
            df_raw = pd.read_csv(io.BytesIO(conteudo), header=None)
            return cls._detectar_e_ajustar_cabecalho(df_raw)
        else:
            raise ValueError(f"Formato de arquivo '{ext}' nÃ£o suportado. Envie uma Imagem (.png, .jpg, .jpeg), PDF (.pdf), Excel (.xlsx, .xls) ou CSV (.csv).")

    @classmethod
    def processar_reajuste(
        cls,
        arquivo_stream,
        nome_arquivo: str,
        modalidade: str = '12_meses',
        margem_adicional: float = 0.0,
        regra_arredondamento: str = 'padrao',
        coluna_preco_selecionada: str = None,
        taxa_manual: float = None,
        mes_vigencia: str = None
    ) -> dict:
        """
        Processa o reajuste de preÃ§os de acordo com o IPCA e gera arquivos de saÃ­da em PDF e Excel.
        """
        cls._validar_mes_vigencia(mes_vigencia)
        t_inicio = time.perf_counter()

        t_ocr_inicio = time.perf_counter()
        df = cls.ler_arquivo(arquivo_stream, nome_arquivo)
        tempo_ocr = time.perf_counter() - t_ocr_inicio

        if df.empty:
            raise ValueError("O documento ou imagem enviada nÃ£o contÃ©m dados legÃ­veis.")

        colunas = cls.deduplicar_colunas(df.columns)
        df.columns = colunas

        # IdentificaÃ§Ã£o inteligente de colunas
        mapeamento = cls.identificar_colunas(colunas)

        coluna_preco = coluna_preco_selecionada if coluna_preco_selecionada and coluna_preco_selecionada in colunas else mapeamento['preco']
        coluna_produto = mapeamento['produto']
        coluna_codigo = mapeamento['codigo']
        coluna_categoria = mapeamento['categoria']

        if not coluna_preco:
            raise ValueError(
                f"NÃ£o foi possÃ­vel identificar a coluna de preÃ§o no documento. "
                f"Colunas encontradas: {', '.join(colunas)}. "
                f"Certifique-se de que hÃ¡ uma coluna com preÃ§os ou valores dos itens."
            )

        t_bacen_inicio = time.perf_counter()
        dados_indice = BacenService.calcular_indice_para_reajuste(modalidade, custom_taxa=taxa_manual)
        tempo_bacen = time.perf_counter() - t_bacen_inicio
        taxa_ipca = dados_indice['percentual']

        fator_ipca = 1.0 + (taxa_ipca / 100.0)
        fator_margem = 1.0 + (margem_adicional / 100.0)
        fator_total = fator_ipca * fator_margem
        variacao_total_pct = round((fator_total - 1.0) * 100.0, 2)

        linhas_resultado = []
        precos_originais = []
        precos_reajustados = []
        qtd_ignoradas = 0

        df_copia = df.copy()
        novos_precos_col = []
        variacoes_rs_col = []
        variacoes_pct_col = []

        for idx, row in df.iterrows():
            raw_preco = row[coluna_preco]
            preco_orig = cls.converter_valor_monetario(raw_preco)

            raw_prod = row[coluna_produto] if coluna_produto else None
            produto_nome = cls._obter_escalar(raw_prod, default=f"Item {idx + 1}")

            raw_cod = row[coluna_codigo] if coluna_codigo else None
            codigo = cls._obter_escalar(raw_cod, default="")

            raw_cat = row[coluna_categoria] if coluna_categoria else None
            categoria = cls._obter_escalar(raw_cat, default="")

            if preco_orig is None or preco_orig <= 0:
                qtd_ignoradas += 1
                novos_precos_col.append(raw_preco if (raw_preco is not None and not isinstance(raw_preco, pd.Series)) else '')
                variacoes_rs_col.append(0.0)
                variacoes_pct_col.append(0.0)
                continue

            novo_preco_bruto = preco_orig * fator_total
            novo_preco = cls.aplicar_regra_arredondamento(novo_preco_bruto, regra_arredondamento)
            dif_rs = round(novo_preco - preco_orig, 2)
            dif_pct = round(((novo_preco / preco_orig) - 1.0) * 100.0, 2)

            precos_originais.append(preco_orig)
            precos_reajustados.append(novo_preco)

            novos_precos_col.append(novo_preco)
            variacoes_rs_col.append(dif_rs)
            variacoes_pct_col.append(dif_pct)

            linhas_resultado.append({
                'linha': idx + 1,
                'codigo': codigo,
                'produto': produto_nome,
                'categoria': categoria,
                'preco_original': preco_orig,
                'preco_original_fmt': f"R$ {preco_orig:,.2f}".replace(',', 'X').replace('.', ',').replace('X', '.'),
                'novo_preco': novo_preco,
                'novo_preco_fmt': f"R$ {novo_preco:,.2f}".replace(',', 'X').replace('.', ',').replace('X', '.'),
                'diferenca_rs': dif_rs,
                'diferenca_rs_fmt': f"+R$ {dif_rs:,.2f}".replace(',', 'X').replace('.', ',').replace('X', '.') if dif_rs >= 0 else f"-R$ {abs(dif_rs):,.2f}".replace(',', 'X').replace('.', ',').replace('X', '.'),
                'diferenca_pct': dif_pct,
                'diferenca_pct_fmt': f"{dif_pct:+.2f}%"
            })

        df_copia['PreÃ§o Reajustado (IPCA)'] = novos_precos_col
        df_copia['Aumento (R$)'] = variacoes_rs_col
        df_copia['VariaÃ§Ã£o (%)'] = variacoes_pct_col

        total_produtos = len(precos_originais)
        soma_original = sum(precos_originais)
        soma_reajustada = sum(precos_reajustados)
        aumento_total_rs = soma_reajustada - soma_original

        kpis = {
            'total_produtos': total_produtos,
            'total_linhas': len(df),
            'linhas_ignoradas': qtd_ignoradas,
            'soma_original': round(soma_original, 2),
            'soma_reajustada': round(soma_reajustada, 2),
            'aumento_total_rs': round(aumento_total_rs, 2),
            'variacao_media_pct': variacao_total_pct,
            'taxa_ipca': taxa_ipca,
            'margem_adicional': margem_adicional,
            'descricao_indice': dados_indice['descricao'],
            'regra_arredondamento': regra_arredondamento,
            'coluna_preco_utilizada': coluna_preco,
            'colunas_disponiveis': colunas
        }

        # GeraÃ§Ã£o do arquivo PDF no formato espelho comparativo solicitado pelo usuÃ¡rio
        t_pdf_inicio = time.perf_counter()
        pdf_bytes = cls.gerar_pdf_reajustado(
            linhas_resultado=linhas_resultado,
            kpis=kpis,
            mes_vigencia=mes_vigencia
        )
        tempo_pdf = time.perf_counter() - t_pdf_inicio

        t_excel_inicio = time.perf_counter()
        excel_bytes = cls._gerar_excel_formatado(
            df_copia,
            coluna_preco,
            taxa_ipca,
            margem_adicional,
            dados_indice['descricao']
        )
        tempo_excel = time.perf_counter() - t_excel_inicio

        tempo_total = time.perf_counter() - t_inicio
        logger.info(
            "REAJUSTE | Tempos por etapa: ocr=%.2fs | bacen=%.2fs | pdf=%.2fs | excel=%.2fs | total=%.2fs",
            tempo_ocr, tempo_bacen, tempo_pdf, tempo_excel, tempo_total
        )

        timestamp_nome = datetime.now().strftime('%Y%m%d_%H%M')

        return {
            'sucesso': True,
            'kpis': kpis,
            'linhas_previa': linhas_resultado[:100],
            'pdf_bytes': pdf_bytes,
            'excel_bytes': excel_bytes,
            'nome_arquivo_pdf': f"produtos_reajustados_ipca_{timestamp_nome}.pdf",
            'nome_arquivo_excel': f"produtos_reajustados_ipca_{timestamp_nome}.xlsx"
        }

    @classmethod

    @classmethod

        provider = cls.get_provider()
        if provider == "texto":
            return True
        try:
            response = requests.get(f"{cls.OLLAMA_HOST}/api/tags", timeout=2)
            response.raise_for_status()

            nomes = {m.get("name", "") for m in response.json().get("models", [])}
            candidatos = {cls.OLLAMA_MODEL}
            if ":" not in cls.OLLAMA_MODEL:
                candidatos.add(f"{cls.OLLAMA_MODEL}:latest")

            if nomes & candidatos:
                return True

            logger.warning("Modelo '%s' nÃ£o encontrado no Ollama.", cls.OLLAMA_MODEL)
            return False
        except requests.RequestException as e:
            logger.warning("Ollama nÃ£o estÃ¡ disponÃ­vel: %s", e)
            return False
        except Exception as e:
            logger.warning("Erro ao verificar Ollama: %s", e)

    @classmethod
    def _processar_com_extrator_texto(
        cls,
        file_bytes: bytes,
        mimetype: str,
        hoje_str: str,
        categorias_existentes: list[str],
    ) -> dict[str, Any]:
        """
        Processa documentos PDF diretamente via extrator_nota.py sem IA/GPU.
        Se falhar ou nÃ£o conseguir identificar valores, retorna estrutura vazia para preenchimento manual.
        """
        import time as _time
        from concurrent.futures import ThreadPoolExecutor, TimeoutError as _FutTimeout

        TIMEOUT_S = min(int(os.getenv("EXTRACAO_TIMEOUT", "90")), 95)
        _t0 = _time.perf_counter()

        from app.services.planilha_produto_service import OCROcupadoError

        try:
            import extrator_nota

            logger.info(
                "EXTRATOR | inÃ­cio | MIME: %s | Bytes: %d | Timeout: %ds",
                mimetype, len(file_bytes), TIMEOUT_S
            )

            def _chamar_extrator():
                if mimetype.endswith("pdf"):
                    return extrator_nota.extrair_dados_pdf(file_bytes)
                elif mimetype.startswith("image/") and os.getenv("OCR_IMAGEM", "1") == "1":
                    logger.info("EXTRATOR | imagem recebida (%s). Iniciando OCR.", mimetype)
                    return extrator_nota.extrair_dados_imagem_ocr(file_bytes)
                return None  # tipo nÃ£o suportado â€” sinaliza abaixo

            if not (mimetype.endswith("pdf") or
                    (mimetype.startswith("image/") and os.getenv("OCR_IMAGEM", "1") == "1")):
                logger.info("EXTRATOR | tipo nÃ£o suportado: %s", mimetype)
                return cls._gerar_resultado_manual(
                    f"Formato '{mimetype}' nÃ£o suportado pelo extrator. Preencha os campos manualmente.",
                    hoje_str,
                )

            
            # Executa em thread separada para limitar a espera do worker.
            executor = ThreadPoolExecutor(max_workers=1)
            futuro = executor.submit(_chamar_extrator)

            try:
                dados_extrator = futuro.result(timeout=TIMEOUT_S)

            except _FutTimeout:
                logger.error(
                    "EXTRATOR | TIMEOUT apÃ³s %ds | MIME: %s | Bytes: %d",
                    TIMEOUT_S,
                    mimetype,
                    len(file_bytes)
                )

                futuro.cancel()

                return cls._gerar_resultado_manual(
                    f"Tempo limite de extraÃ§Ã£o excedido ({TIMEOUT_S}s). "
                    "Preencha os dados manualmente.",
                    hoje_str,
                )

            finally:
                # NÃ£o aguarda uma tarefa que ultrapassou o limite.
                executor.shutdown(wait=False, cancel_futures=True)

            if dados_extrator is None:
                return cls._gerar_resultado_manual(
                    f"Formato '{mimetype}' nÃ£o suportado pelo extrator. Preencha os campos manualmente.",
                    hoje_str,
                )

            logger.info(
                "EXTRATOR | extraÃ§Ã£o concluÃ­da em %.1fs | fornecedor=%r | valor=%.2f | parcelas=%d",
                _time.perf_counter() - _t0,
                dados_extrator.get("fornecedor", ""),
                float(dados_extrator.get("valor_total") or 0),
                int(dados_extrator.get("parcelas") or 1),
            )

            # Converte o retorno do extrator_nota para o formato esperado pela tela de revisÃ£o
            dados_convertidos = cls._converter_extrator_para_revisao(dados_extrator, hoje_str, categorias_existentes)

            logger.info(
                "EXTRATOR | conversÃ£o concluÃ­da | forma_pagamento=%r | total=%.2f",
                dados_convertidos.get("forma_pagamento"),
                dados_convertidos.get("valor_total", 0),
            )

            if dados_convertidos["valor_total"] == 0.0 and not dados_convertidos["itens"]:
                return cls._gerar_resultado_manual(
                    "NÃ£o foi possÃ­vel identificar os dados automaticamente no documento. Por favor, preencha manualmente.",
                    hoje_str,
                )

            dados_convertidos["ia_sucesso"] = True
            dados_convertidos["modo_demo"] = False
            return dados_convertidos

        except OCROcupadoError:
            # Propaga a ocupacao para o chamador.
            raise

        except Exception as e:
            # Loga com traceback completo, sem engolir exceÃ§Ãµes
            logger.exception(
                "EXTRATOR | ERRO apÃ³s %.1fs | MIME: %s | %s",
                _time.perf_counter() - _t0, mimetype, e
            )
            return cls._gerar_resultado_manual(
                f"Erro na extraÃ§Ã£o automÃ¡tica ({e}). Por favor, preencha os dados manualmente.",
                hoje_str,
            )

        }

    @classmethod
    def _otimizar_imagem(cls, img_bytes: bytes) -> bytes:
        """Corrige rotaÃ§Ã£o, reduz resoluÃ§Ã£o e converte para JPEG."""
        img = Image.open(BytesIO(img_bytes))
        img = ImageOps.exif_transpose(img).convert("RGB")
        img.thumbnail((cls.IMG_LADO_MAX, cls.IMG_LADO_MAX))
        buf = BytesIO()
        img.save(buf, "JPEG", quality=88)
        return buf.getvalue()

    @classmethod
    def _preparar_imagens(cls, file_bytes: bytes, mimetype: str) -> list[bytes]:
        if mimetype.startswith("image/"):
            return [cls._otimizar_imagem(file_bytes)]

        if mimetype == "application/pdf":
            try:
                import fitz  # PyMuPDF
            except ImportError:
                raise RuntimeError("PyMuPDF nÃ£o estÃ¡ instalado. Execute: pip install pymupdf")

            documento = fitz.open(stream=file_bytes, filetype="pdf")
            try:
                total = len(documento)
                limite = min(total, cls.MAX_PAGINAS_PDF)
                imagens = []
                for n in range(limite):
                    pixmap = documento.load_page(n).get_pixmap(
                        matrix=fitz.Matrix(2, 2), alpha=False
                    )
                    imagens.append(cls._otimizar_imagem(pixmap.tobytes("png")))
            finally:
                documento.close()

            if not imagens:
                raise ValueError("O PDF nÃ£o possui pÃ¡ginas legÃ­veis.")

            if total > limite:
                logger.warning(
                    "PDF com %s pÃ¡ginas; somente as primeiras %s foram processadas.",
                    total,
                    limite,
                )
            return imagens

        raise ValueError(f"Tipo de arquivo nÃ£o suportado: {mimetype}")

    def _extrair_json(texto: str) -> dict[str, Any]:
        texto = texto.strip()
        texto = re.sub(r"^```json\s*", "", texto, flags=re.IGNORECASE)
        texto = re.sub(r"^```\s*", "", texto)
        texto = re.sub(r"\s*```$", "", texto).strip()

        try:
            resultado = json.loads(texto)
            if isinstance(resultado, dict):
                return resultado
        except json.JSONDecodeError:
            pass

        inicio, fim = texto.find("{"), texto.rfind("}")
        if inicio == -1 or fim <= inicio:
            raise ValueError("O modelo nÃ£o retornou um JSON vÃ¡lido.")

        try:
            resultado = json.loads(texto[inicio : fim + 1])
        except json.JSONDecodeError as e:
            raise ValueError(f"JSON retornado pelo modelo Ã© invÃ¡lido: {e}")

        if not isinstance(resultado, dict):
            raise ValueError("A resposta JSON nÃ£o Ã© um objeto.")
        return resultado

    # ------------------------------------------------------------------
    # NormalizaÃ§Ã£o
    # ------------------------------------------------------------------
    @staticmethod
    def _para_float(valor: Any) -> float:
        if isinstance(valor, bool):
            return 0.0
        if isinstance(valor, (int, float)):
            return float(valor)
        if isinstance(valor, str):
            texto = re.sub(r"[^\d,.\-]", "", valor)
            if not texto:
                return 0.0
            if "," in texto and "." in texto:
                if texto.rfind(",") > texto.rfind("."):
                    texto = texto.replace(".", "").replace(",", ".")
                else:
                    texto = texto.replace(",", "")
            elif "," in texto:
                texto = texto.replace(",", ".")
            try:
                return float(texto)
            except ValueError:
                return 0.0
        return 0.0

    @staticmethod
    def _para_texto(valor: Any) -> str:
        return "" if valor is None else str(valor).strip()

    @classmethod
    def _normalizar_data(cls, valor: Any, hoje_str: str) -> str:
        texto = cls._para_texto(valor)

        if re.fullmatch(r"\d{4}-\d{2}-\d{2}", texto):
            try:
                date.fromisoformat(texto)
                return texto
            except ValueError:
                return hoje_str

        m = re.fullmatch(r"(\d{2})/(\d{2})/(\d{4})", texto)
        if m:
            try:
                return date(int(m.group(3)), int(m.group(2)), int(m.group(1))).isoformat()
            except ValueError:
                return hoje_str

        return hoje_str


    @classmethod
    def _detectar_parcelamento_texto(
        cls,
        texto: str
    ) -> tuple[bool, int, float]:
        """Detecta parcelamentos em textos de comprovantes."""

        if not texto:
            return False, 1, 0.0

        texto = texto.strip()

        # Exemplo: PARC. 1/6, PARCELA 1/6, 01/06
        padrao_fracao = (
            r"(?:parc(?:ela|elas)?\.?\s*)?"
            r"\d{1,2}\s*/\s*(\d{1,2})"
        )

        m = re.search(padrao_fracao, texto, re.IGNORECASE)

        if m:
            num = int(m.group(1))

            if 2 <= num <= 36:
                return True, num, 0.0

        # Exemplo: 3x de R$ 50,00
        m = re.search(
            r"(\d{1,2})\s*[xX]\s*(?:de\s*)?R?\$?\s*([\d.,]+)",
            texto
        )

        if m:
            num = int(m.group(1))

            if 2 <= num <= 36:
                val_str = m.group(2)

                if "," in val_str and "." in val_str:
                    if val_str.rfind(",") > val_str.rfind("."):
                        val_str = val_str.replace(".", "").replace(",", ".")
                    else:
                        val_str = val_str.replace(",", "")
                elif "," in val_str:
                    val_str = val_str.replace(",", ".")

                try:
                    valor = float(val_str)
                except ValueError:
                    valor = 0.0

                return True, num, valor

        # Exemplo: parcelado em 6, parcelas: 6, parc. 6
        m = re.search(
            r"(?:parcel(?:as?|ado\s*em)|parc\.?)"
            r"\s*:?\s*(\d{1,2})(?!\s*/)",
            texto,
            re.IGNORECASE
        )

        if m:
            num = int(m.group(1))

            if 2 <= num <= 36:
                return True, num, 0.0

        # Exemplo: 6 parcelas
        m = re.search(
            r"(\d{1,2})\s*parcelas",
            texto,
            re.IGNORECASE
        )

        if m:
            num = int(m.group(1))

            if 2 <= num <= 36:
                return True, num, 0.0

        return False, 1, 0.0


    @classmethod
    def _normalizar_resultado(cls, dados: dict[str, Any], hoje_str: str) -> dict[str, Any]:
        tipo_transacao = cls._para_texto(dados.get("tipo_transacao")).lower()
        if tipo_transacao not in cls.TIPOS_TRANSACAO:
            tipo_transacao = "despesa"

        forma_pagamento = cls._para_texto(dados.get("forma_pagamento"))
        if forma_pagamento not in cls.FORMAS_PAGAMENTO:
            forma_pagamento = ""

        tipo_receita = cls._para_texto(dados.get("tipo_receita"))
        if tipo_receita not in cls.TIPOS_RECEITA:
            tipo_receita = "Outros"

        itens_brutos = dados.get("itens")
        if not isinstance(itens_brutos, list):
            itens_brutos = []

        itens = []
        for item in itens_brutos:
            if not isinstance(item, dict):
                continue
            itens.append(
                {
                    "descricao": cls._para_texto(item.get("descricao")),
                    "valor": round(cls._para_float(item.get("valor")), 2),
                    "categoria_sugerida": cls._para_texto(item.get("categoria_sugerida")),
                }
            )

        valor_total = round(cls._para_float(dados.get("valor_total")), 2)
        observacoes = cls._para_texto(dados.get("observacoes"))

        # ExtraÃ§Ã£o de dados de parcelamento
        
        # Converte booleanos sem interpretar a string "false" como True.
        valor_booleano = dados.get("is_parcelado", False)

        if isinstance(valor_booleano, bool):
            is_parcelado = valor_booleano
        elif isinstance(valor_booleano, str):
            is_parcelado = valor_booleano.strip().lower() in (
                "true", "1", "sim", "yes"
            )
        else:
            is_parcelado = valor_booleano == 1

        numero_parcelas = int(
            cls._para_float(dados.get("numero_parcelas") or 1)
        )

        numero_parcelas = max(1, min(numero_parcelas, 36))

        valor_parcela = round(
            cls._para_float(dados.get("valor_parcela")),
            2
        )


        # Se a IA nÃ£o sinalizou no schema mas o texto tem evidÃªncia de parcelamento
        if not is_parcelado or numero_parcelas <= 1:
            texto_busca = f"{observacoes} {dados.get('estabelecimento', '')} " + " ".join(i.get('descricao', '') for i in itens)
            det, num_det, val_det = cls._detectar_parcelamento_texto(texto_busca)
            if det:
                is_parcelado = True
                numero_parcelas = num_det
                if val_det > 0 and valor_parcela == 0:
                    valor_parcela = val_det

        if numero_parcelas > 1:
            is_parcelado = True
            if not forma_pagamento or forma_pagamento not in ["CartÃ£o de CrÃ©dito"]:
                forma_pagamento = "CartÃ£o de CrÃ©dito"
            if valor_parcela == 0 and valor_total > 0:
                valor_parcela = round(valor_total / numero_parcelas, 2)
        else:
            numero_parcelas = 1

        if itens:
            soma = round(sum(i["valor"] for i in itens), 2)
            if soma == 0.0 and valor_total > 0:
                # Itens sem valor no documento: lanÃ§a um Ãºnico item com o total.
                primeiro = itens[0]
                itens = [
                    {
                        "descricao": primeiro["descricao"] or "Total do documento",
                        "valor": valor_total,
                        "categoria_sugerida": primeiro["categoria_sugerida"],
                    }
                ]
                aviso = (
                    "Os itens do documento estavam com valor 0,00; foi lanÃ§ado "
                    "um Ãºnico item com o valor total."
                )
                observacoes = f"{observacoes}\n{aviso}".strip()
            elif valor_total == 0.0 and soma > 0:
                valor_total = soma
            elif abs(soma - valor_total) > 0.05:
                aviso = (
                    f"ATENÃ‡ÃƒO: a soma dos itens (R$ {soma:.2f}) difere do "
                    f"total (R$ {valor_total:.2f}). Confira antes de salvar."
                )
                observacoes = f"{observacoes}\n{aviso}".strip()

        return {
            "tipo_transacao": tipo_transacao,
            "data": cls._normalizar_data(dados.get("data"), hoje_str),
            "estabelecimento": cls._para_texto(dados.get("estabelecimento")),
            "forma_pagamento": forma_pagamento,
            "tipo_receita": tipo_receita,
            "valor_total": valor_total,
            "is_parcelado": is_parcelado,
            "numero_parcelas": numero_parcelas,
            "valor_parcela": valor_parcela,
            "itens": itens,
            "observacoes": observacoes,
        }

    # ------------------------------------------------------------------
    # Erro
    # ------------------------------------------------------------------

    @classmethod
    def _gerar_resultado_manual(cls, mensagem: str, hoje_str: str) -> dict[str, Any]:
        return cls._gerar_resultado_erro(mensagem, hoje_str)

    @staticmethod
    def _gerar_resultado_erro(mensagem: str, hoje_str: str) -> dict[str, Any]:

def _ensure_tmp_dir() -> None:

def allowed_file(filename: str) -> bool:
    return (
        bool(filename)

def _preview_path(token: str, ext: str) -> str:
    safe_token = "".join(
        c for c in token if c.isalnum() or c == "-"
    )
    safe_ext = "".join(c for c in ext if c.isalnum())

    if not safe_token or safe_ext not in ALLOWED_EXTENSIONS:
        raise ValueError("Identificador de prÃ©via invÃ¡lido.")

    return os.path.join(TMP_IA_DIR, f"{safe_token}.{safe_ext}")

def _converter_valor(valor) -> Decimal | None:
    """Converte valor monetÃ¡rio brasileiro ou decimal para centavos."""
    if valor is None:
        return None

    texto = str(valor).strip().replace("R$", "").replace(" ", "")

    if not texto:
        return None

    if "," in texto and "." in texto:
        if texto.rfind(",") > texto.rfind("."):
            texto = texto.replace(".", "").replace(",", ".")
        else:
            texto = texto.replace(",", "")
    elif "," in texto:
        texto = texto.replace(".", "").replace(",", ".")

    try:
        resultado = Decimal(texto).quantize(Decimal("0.01"))
        return resultado if resultado.is_finite() else None
    except (InvalidOperation, ValueError):
        return None

def _limpar_jobs_antigos() -> None:
    limite = time.time() - JOB_TTL_SEGUNDOS

    with _JOBS_LOCK:
        antigos = [
            job_id
            for job_id, job in _JOBS.items()
            if job.get("criado", 0) < limite
            and job.get("status") != "processando"
        ]

        for job_id in antigos:
            _JOBS.pop(job_id, None)


def _usuario_tem_job_ativo(id_usuario) -> bool:
    with _JOBS_LOCK:
        return any(
            job.get("usuario") == id_usuario
            and job.get("status") == "processando"
            for job in _JOBS.values()
        )


def _executar_job(
    job_id: str,
    file_bytes: bytes,
    mimetype: str,
    ext: str,
    filename: str,
    origem: str,
    cat_nomes: list[str],
) -> None:
    """
    Roda fora da requisiÃ§Ã£o. NÃ£o pode usar session, request nem flash:
    o resultado Ã© entregue pela rota /ia/status.
    """
    try:
        resultado = AiFiscalService.interpretar_documento(
            file_bytes,
            mimetype,
            cat_nomes
        )

        if not isinstance(resultado, dict):
            raise ValueError("O serviÃ§o de leitura retornou dados invÃ¡lidos.")

        _ensure_tmp_dir()

        token = str(uuid.uuid4())

        with open(_preview_path(token, ext), "wb") as arquivo:
            arquivo.write(file_bytes)

        novo_estado = {
            "status": "ok",
            "preview": {
                "dados": resultado,
                "file_token": token,
                "ext": ext,
                "mimetype": mimetype,
                "filename": os.path.basename(filename),
                "origem": origem,
            },
        }

    except Exception as erro:
        from app.services.planilha_produto_service import OCROcupadoError

        if isinstance(erro, OCROcupadoError):
            logger.warning(
                "OCR ocupado | Job fiscal %s recusado: %s",
                job_id,
                erro
            )
            mensagem = str(erro)
        else:
            logger.exception(
                "Erro ao processar documento fiscal (job %s).", job_id
            )
            mensagem = "Nao foi possivel processar o arquivo enviado."

        novo_estado = {
            "status": "erro",
            "msg": mensagem,
        }

    with _JOBS_LOCK:
        if job_id in _JOBS:
            _JOBS[job_id].update(novo_estado)

@ia_fiscal_bp.route("/processar", methods=["POST"])
@login_required
def processar():
    """
    Valida o arquivo e inicia a leitura em segundo plano.
    Responde na hora com um job_id; a pÃ¡gina consulta /ia/status/<job_id>.
    """
    file = request.files.get("documento")

    if not file or not file.filename:
        return _json_erro("Selecione um arquivo de imagem ou PDF.")

    if not allowed_file(file.filename):
        return _json_erro(
            "Formato nÃ£o suportado. Envie PNG, JPG, JPEG, WEBP ou PDF."
        )

    ext = file.filename.rsplit(".", 1)[1].lower()
    file_bytes = file.read()

    logger.info(
        "UPLOAD | MIME: %s | ExtensÃ£o: %s | Bytes: %d",
        file.mimetype,
        ext,
        len(file_bytes)
    )

    if not file_bytes:
        return _json_erro("O arquivo enviado estÃ¡ vazio.")

    if len(file_bytes) > 16 * 1024 * 1024:
        return _json_erro("Arquivo muito grande (mÃ¡ximo 16 MB).")

    mimetypes = {
        "pdf": "application/pdf",
        "jpg": "image/jpeg",
        "jpeg": "image/jpeg",
        "png": "image/png",
        "webp": "image/webp",
    }
    mimetype = mimetypes[ext]

    id_usuario = session.get("id_usuario")

    _limpar_jobs_antigos()

    # Evita empilhar leituras: o OCR Ã© compartilhado e novas tentativas
    # sÃ³ ficariam esperando na fila.
    if _usuario_tem_job_ativo(id_usuario):
        return _json_erro(
            "JÃ¡ existe uma leitura em andamento. "
            "Aguarde alguns segundos e tente novamente.",
            409
        )

    try:
        cat_nomes = [
            c.nome_categoria for c in CategoriaService.listar_todas()
        ]
    except Exception:
        logger.exception("Erro ao carregar categorias.")
        return _json_erro(
            "NÃ£o foi possÃ­vel carregar as categorias.", 500
        )

    job_id = uuid.uuid4().hex

    with _JOBS_LOCK:
        _JOBS[job_id] = {
            "status": "processando",
            "criado": time.time(),
            "usuario": id_usuario,
            "entregue": False,
        }

    try:
        threading.Thread(
            target=_executar_job,
            args=(
                job_id,
                file_bytes,
                mimetype,
                ext,
                file.filename,
                request.form.get("origem", "despesa"),
                cat_nomes,
            ),
            daemon=True,
        ).start()
    except Exception:
        logger.exception(
            "Nao foi possivel iniciar a thread do job fiscal %s.",
            job_id
        )
        with _JOBS_LOCK:
            _JOBS.pop(job_id, None)
        return _json_erro(
            "Nao foi possivel iniciar o processamento. Tente novamente.",
            500
        )

    return jsonify(status="processando", job_id=job_id), 202


@ia_fiscal_bp.route("/status/<job_id>")
@login_required
def status(job_id):
    """Informa o andamento da leitura e, quando pronta, grava a prÃ©via na sessÃ£o."""
    with _JOBS_LOCK:
        job = _JOBS.get(job_id)

        if job is not None:
            job = dict(job)  # cÃ³pia para ler fora do bloqueio

    if not job or job["usuario"] != session.get("id_usuario"):
        return jsonify(status="inexistente"), 404

    if job["status"] == "processando":
        return jsonify(status="processando")

    if job["status"] == "erro":
        return jsonify(status="erro", msg=job.get("msg"))

    # Pronto. A entrega Ã© idempotente: se o navegador perder a resposta
    # (o iOS suspende a pÃ¡gina com frequÃªncia) e perguntar de novo,
    # recebe o mesmo redirecionamento.
    if not job["entregue"]:
        _limpar_preview_temporario()
        session["ia_preview"] = job["preview"]

        with _JOBS_LOCK:
            if job_id in _JOBS:
                _JOBS[job_id]["entregue"] = True

        dados = job["preview"]["dados"]

        if dados.get("ia_sucesso"):
            flash(
                "Documento interpretado! Revise os dados antes de confirmar.",
                "success"
            )
        else:
            erro_msg = dados.get(
                "erro_msg",
                "NÃ£o foi possÃ­vel extrair os dados automaticamente."
            )
            flash(f"âš ï¸ {erro_msg}", "warning")

    elif not session.get("ia_preview"):
        session["ia_preview"] = job["preview"]

    return jsonify(status="ok", redirect=url_for("ia_fiscal.revisar"))

def _imagem_bytes_para_texto_ocr(image_bytes: bytes) -> str:
    """
    Recebe bytes de uma imagem e retorna o texto extraÃ­do
    via RapidOCR, registrando o tempo de cada etapa.
    """

    import numpy as np
    import os

    from PIL import Image, ImageOps

    inicio_total = time.perf_counter()

    logger.info(
        "OCR | Iniciando processamento | Tamanho: %.2f MB",
        len(image_bytes) / (1024 * 1024)
    )

    # ETAPA 1: Obter o motor OCR
    inicio = time.perf_counter()

    try:
        _obter_ocr_engine()

    except Exception:
        logger.exception("OCR | Erro ao obter/inicializar o motor")
        raise

    tempo_motor = time.perf_counter() - inicio

    logger.info(
        "OCR | Motor obtido em %.2f segundos",
        tempo_motor
    )

    # ETAPA 2: Abrir e preparar a imagem
    inicio = time.perf_counter()

    try:
        lado_max = int(os.getenv("OCR_MAX_LADO", "1280"))
        alvo = int(os.getenv("OCR_ALVO", "1280"))

        with Image.open(io.BytesIO(image_bytes)) as imagem_original:
            largura_original, altura_original = imagem_original.size

            # JPEG: o decodificador jÃ¡ entrega a imagem reduzida (bem menos RAM).
            if imagem_original.format == "JPEG":
                imagem_original.draft("RGB", (lado_max, lado_max))

            img = ImageOps.exif_transpose(imagem_original).convert("RGB")

        # Limita o tamanho mÃ¡ximo da imagem
        img.thumbnail((lado_max, lado_max))

        # Amplia imagens pequenas somente quando necessÃ¡rio
        fator = min(3, max(1, round(alvo / max(img.size))))

        if fator > 1:
            img = img.resize(
                (img.width * fator, img.height * fator),
                Image.LANCZOS
            )

        largura_final, altura_final = img.size

        imagem_np = np.array(img)

        logger.info(
            "OCR | Imagem preparada | Original: %sx%s | Final: %sx%s",
            largura_original,
            altura_original,
            largura_final,
            altura_final
        )

    except Exception:
        logger.exception("OCR | Erro ao preparar a imagem")
        raise

    tempo_preparo = time.perf_counter() - inicio

    logger.info(
        "OCR | PreparaÃ§Ã£o concluÃ­da em %.2f segundos",
        tempo_preparo
    )

    # ETAPA 3: Executar o RapidOCR
    inicio = time.perf_counter()

    try:
        logger.info(
            "OCR | Aguardando execuÃ§Ã£o compartilhada do RapidOCR"
        )

        res, _ = _executar_ocr(imagem_np)

        tempo_inferencia = time.perf_counter() - inicio

        quantidade = len(res) if res else 0

        logger.info(
            "OCR | InferÃªncia concluÃ­da em %.2f segundos | "
            "Caixas detectadas: %d",
            tempo_inferencia,
            quantidade
        )

    except Exception:
        logger.exception("OCR | Erro durante a inferÃªncia")
        raise

    if not res:
        logger.warning("OCR | Nenhum texto detectado")

        logger.info(
            "OCR | Processamento total: %.2f segundos",
            time.perf_counter() - inicio_total
        )

        return ""

    # ETAPA 4: Organizar os textos por linha
    inicio = time.perf_counter()

    caixas = []

    for box, texto, *_ in res:
        texto = str(texto).strip()

        if not texto:
            continue

        ys = [p[1] for p in box]
        xs = [p[0] for p in box]

        caixas.append({
            "texto": texto,
            "y": (min(ys) + max(ys)) / 2.0,
            "x": min(xs),
            "h": max(ys) - min(ys)
        })

    if not caixas:
        logger.warning("OCR | Nenhuma caixa de texto utilizÃ¡vel")

        logger.info(
            "OCR | Processamento total: %.2f segundos",
            time.perf_counter() - inicio_total
        )

        return ""

    altura_mediana = sorted(
        c["h"] for c in caixas
    )[len(caixas) // 2]

    tolerancia = max(4.0, 0.6 * altura_mediana)

    caixas.sort(key=lambda c: c["y"])

    linhas_agrupadas = []

    for c in caixas:
        if (
            linhas_agrupadas
            and abs(linhas_agrupadas[-1]["y"] - c["y"]) <= tolerancia
        ):
            linhas_agrupadas[-1]["itens"].append(c)

            linhas_agrupadas[-1]["y"] += (
                c["y"] - linhas_agrupadas[-1]["y"]
            ) / len(linhas_agrupadas[-1]["itens"])

        else:
            linhas_agrupadas.append({
                "y": c["y"],
                "itens": [c]
            })

    linhas_texto = []

    for linha in linhas_agrupadas:
        texto_linha = " ".join(
            i["texto"]
            for i in sorted(
                linha["itens"],
                key=lambda i: i["x"]
            )
        )

        linhas_texto.append(texto_linha.strip())

    texto_final = "\n".join(linhas_texto)

    tempo_organizacao = time.perf_counter() - inicio
    tempo_total = time.perf_counter() - inicio_total

    logger.info(
        "OCR | OrganizaÃ§Ã£o concluÃ­da em %.2f segundos",
        tempo_organizacao
    )

    logger.info(
        "OCR | Finalizado | Caracteres: %d | Tempo total: %.2f segundos",
        len(texto_final),
        tempo_total
    )

    return texto_final

def extrair_dados_imagem_ocr(image_bytes: bytes) -> dict[str, Any]:
    """
    Extrai dados de nota fiscal / comprovante a partir de uma imagem
    (JPEG, PNG, WEBP), usando RapidOCR e os mesmos regex de extrair_dados_pdf.

        "EXTRATOR | extrair_dados_imagem_ocr | inÃ­cio | %d bytes",
        len(image_bytes)
    )

    hash_arquivo = hashlib.sha256(image_bytes).hexdigest()
    cached = _obter_ocr_cache(f"nota_{hash_arquivo}")
    if cached is not None:
        logger.info(
            "EXTRATOR | extrair_dados_imagem_ocr | Cache SHA-256 atingido (%s) | %.3fs",
            hash_arquivo[:12], time.perf_counter() - _t0_ocr_fn
        )
        return copy.deepcopy(cached)

    from app.services.planilha_produto_service import (
        _OCR_PIPELINE_LOCK,
        OCROcupadoError,
    )

    if not _OCR_PIPELINE_LOCK.acquire(blocking=False):
        raise OCROcupadoError(
            "O OCR esta processando outro documento. Aguarde alguns instantes e tente novamente."
        )

    try:
        texto_completo = _imagem_bytes_para_texto_ocr(image_bytes)
    finally:
        _OCR_PIPELINE_LOCK.release()

    # Diagn?stico tempor?rio: somente linhas relacionadas ao pagamento.
    for linha_ocr in texto_completo.splitlines():
        if re.search(r"VENDA|CREDITO|CR[E?]DITO|DOC|TED|TEF|TRANSFER", linha_ocr, re.IGNORECASE):
            logger.info("OCR | Linha de pagamento: %s", linha_ocr[:180])

    if not texto_completo.strip():
        logger.warning(
            "EXTRATOR | extrair_dados_imagem_ocr | OCR sem texto | %.1fs",
            time.perf_counter() - _t0_ocr_fn
        )
        return {
            "fornecedor": "",
            "data": "",
            "forma_pagamento": "",
            "parcelas": 1,
            "valor_total": 0.0,
            "itens": [],
        }

    logger.info(
        "EXTRATOR | extrair_dados_imagem_ocr | OCR concluÃ­do | %d chars | %.1fs",
        len(texto_completo), time.perf_counter() - _t0_ocr_fn
    )

    # Reutiliza o pipeline de regex
    resultado = _extrair_de_texto(texto_completo)
    _salvar_ocr_cache(f"nota_{hash_arquivo}", resultado)
    logger.info(
        "EXTRATOR | extrair_dados_imagem_ocr | fim | total=%.2f | %.1fs",
        float(resultado.get("valor_total") or 0),
        time.perf_counter() - _t0_ocr_fn,
    )
    return resultado

def extrair_dados_pdf(
    file_source: bytes | io.BytesIO | str
) -> dict[str, Any]:
    """
    LÃª o texto e tabelas de um documento PDF e devolve os dados
    estruturados da nota fiscal/comprovante.
    """

    if isinstance(file_source, bytes):
        stream = io.BytesIO(file_source)

    elif isinstance(file_source, io.BytesIO):
        stream = file_source

    elif isinstance(file_source, str):
        stream = file_source

    else:
        raise ValueError(
            f"Fonte de arquivo invÃ¡lida: {type(file_source)}"
        )

    texto_completo = ""
    linhas = []

    with pdfplumber.open(stream) as pdf:
        for page in pdf.pages:
            t = page.extract_text() or ""

            texto_completo += t + "\n"

            for line in t.splitlines():
                line_str = line.strip()

                if line_str:
                    linhas.append(line_str)

    if not texto_completo.strip():
        logger.warning(
            "Nenhum texto pÃ´de ser extraÃ­do do PDF "
            "(pode ser imagem digitalizada sem camada de texto)."
        )

        return {
            'fornecedor': '',
            'data': '',
            'forma_pagamento': '',
            'parcelas': 1,
            'valor_total': 0.0,
            'itens': []
        }

    return _extrair_de_texto(texto_completo, linhas)

# PIPELINE PRINCIPAL DE EXTRAÃ‡ÃƒO POR EXPRESSÃ•ES REGULARES
# ============================================================

def _extrair_de_texto(
    texto_completo: str,
    linhas: list[str] | None = None
) -> dict[str, Any]:
    """
    Pipeline principal de extraÃ§Ã£o por regex.

    Recebe o texto bruto (de pdfplumber ou de OCR) e retorna
    o dicionÃ¡rio padronizado.

    Se 'linhas' nÃ£o for fornecida, Ã© gerada a partir de 'texto_completo'.
    """

    if linhas is None:
        linhas = [
            l.strip()
            for l in texto_completo.splitlines()
            if l.strip()
        ]

    # ========================================================
    # 1. IDENTIFICAR FORNECEDOR / ESTABELECIMENTO / BENEFICIÃRIO
    # ========================================================

    fornecedor = ""

    # Comprovantes Pix: prioriza o nome de quem recebeu a transfer?ncia.
    m_recebedor = re.search(
        r"DADOS DE QUEM RECEBEU\s*[:\-]?\s*.*?NOME\s*:\s*([^\n\r]+)",
        texto_completo,
        re.IGNORECASE | re.DOTALL
    )

    if m_recebedor:
        fornecedor = m_recebedor.group(1).strip()


    # Casos DANFE / NF-e:
    # "RECEBEMOS DE <NOME>" ou "EMITENTE: <NOME>" ou "RAZÃƒO SOCIAL"

    m_rec = re.search(
        r"RECEBEMOS\s+DE\s+([^\n\r,]+)",
        texto_completo,
        re.IGNORECASE
    )

    if m_rec:
        fornecedor = m_rec.group(1).strip()

    elif (
        "BeneficiÃ¡rio" in texto_completo
        or "Beneficirio" in texto_completo
    ):
        for i, l in enumerate(linhas):

            if re.search(
                r"Benefici[aÃ¡\uFFFD]rio",
                l,
                re.IGNORECASE
            ):
                if i + 1 < len(linhas):
                    prox = linhas[i + 1]

                    # Limpa CNPJ/CPF se vier na mesma linha
                    prox_clean = re.sub(
                        r"\d{2}\.?\d{3}\.?\d{3}/\d{4}-\d{2}|\d{11,14}",
                        "",
                        prox
                    ).strip()

                    fornecedor = prox_clean.split("  ")[0].strip()
                    break

    if not fornecedor:

        m_emit = re.search(
            r"(?:EMITENTE|RAZ[AÃƒ\uFFFD]O\s+SOCIAL|FORNECEDOR)[:\s]+([^\n\r]+)",
            texto_completo,
            re.IGNORECASE
        )

        if m_emit:
            fornecedor = m_emit.group(1).strip()

        else:
            # Fallback aprimorado: procura padrÃµes estruturados antes de
            # usar a primeira linha genÃ©rica.

            # Tenta "EMPRESA:", "LOJA:", "ESTABELECIMENTO:" no texto
            m_extra = re.search(
                r"(?:EMPRESA|LOJA|ESTABELECIMENTO|CEDENTE|SACADO)[:\s]+([^\n\r,]{3,60})",
                texto_completo,
                re.IGNORECASE
            )
            if m_extra:
                fornecedor = m_extra.group(1).strip()

            else:
                # Procura a linha imediatamente antes de um CNPJ
                # (nome da empresa quase sempre antecede o CNPJ)
                cnpj_match = re.search(
                    r"(\d{2}\.?\d{3}\.?\d{3}/\d{4}-\d{2})",
                    texto_completo
                )
                if cnpj_match:
                    pos = cnpj_match.start()
                    trecho_antes = texto_completo[:pos]
                    linhas_antes = [l.strip() for l in trecho_antes.splitlines() if l.strip()]
                    for cand in reversed(linhas_antes[-4:]):
                        # ignora linhas que sÃ£o sÃ³ nÃºmeros/CNPJ/endereÃ§o
                        if not re.search(
                            r"^\d|RUA|AV\.|AVENIDA|R\.|CEP|FONE|TEL|FAX",
                            cand, re.IGNORECASE
                        ) and len(cand) > 3:
                            fornecedor = cand
                            break

            if not fornecedor:
                # Ãšltimo recurso: primeira linha informativa entre as 7 primeiras
                _skip = re.compile(
                    r"INFORMATIVO|DANFE|DOCUMENTO|NOTA FISCAL|RECIBO"
                    r"|COMPROVANTE|PREFEITURA|SECRETARIA|MUNICIPIO|FATURA"
                    r"|DUPLICATA|CUPOM|TRIBUT",
                    re.IGNORECASE
                )
                for l in linhas[:7]:
                   candidato = l.strip()

                   if (
                       len(candidato) > 3
                       and not _skip.search(candidato)
                       and not re.search(r"\\d{2}[/.-]\\d{2}[/.-]\\d{4}", candidato)
                       and not re.search(r"\\d{1,2}:\\d{2}", candidato)
                       and not re.fullmatch(r"[\\d\\s.,/:-]+", candidato)
                   ):
                       fornecedor = candidato
                       break

    # ========================================================
    # 2. IDENTIFICAR DATA
    # ========================================================

    data_formatada = ""

    # PadrÃ£o DD/MM/AAAA, DD-MM-AAAA ou DD.MM.AAAA

    datas_encontradas = re.findall(
        r"\b(\d{2})[/.-](\d{2})[/.-](\d{4})\b",
        texto_completo
    )

    if datas_encontradas:

        for d, m, a in datas_encontradas:

            try:
                dt = datetime(
                    int(a),
                    int(m),
                    int(d)
                )

                data_formatada = dt.strftime("%Y-%m-%d")
                break

            except ValueError:
                continue

    # ========================================================
    # 3. IDENTIFICAR FORMA DE PAGAMENTO E PARCELAS
    # ========================================================

    # â”€â”€ Forma de pagamento: busca contextual (evita falso-positivo) â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    forma_pagamento = ""
    parcelas = 1

    t_upper = texto_completo.upper()

    # Normaliza acentos para evitar problemas de codifica??o e OCR.
    import unicodedata

    t_norm = "".join(
        caractere
        for caractere in unicodedata.normalize("NFD", t_upper)
        if unicodedata.category(caractere) != "Mn"
    )

    # Prioridade 1: Pix
    if re.search(r"\bPIX\b|PAGAMENTO\s+(?:VIA\s+)?PIX|CHAVE\s+PIX", t_norm):
        forma_pagamento = "Pix"

    # Prioridade 2: Cart?o de cr?dito
    elif re.search(
        r"CARTAO\s+(?:DE\s+)?CREDITO"
        r"|CREDITO\s+EM\s+\d+\s*[Xx]"
        r"|COMPRA\s+CREDITO"
        r"|CREDITO\s+PARCELADO"
        r"|VENDA\s*A?\s*CREDITO",
        t_norm
    ):
        forma_pagamento = "Cart\u00e3o de Cr\u00e9dito"

    # Prioridade 3: Cart?o de d?bito
    elif re.search(
        r"CARTAO\s+(?:DE\s+)?DEBITO"
        r"|DEBITO\s+(?:EM\s+CONTA|AUTOMATICO|CONTA)"
        r"|COMPRA\s+DEBITO",
        t_norm
    ):
        forma_pagamento = "Cart\u00e3o de D\u00e9bito"

    # Prioridade 4: Dinheiro
    elif re.search(
        r"\bDINHEIRO\b|\bESPECIE\b|PAGAMENTO\s+EM\s+DINHEIRO",
        t_norm
    ):
        forma_pagamento = "Dinheiro"

    # Prioridade 5: Transfer?ncia
    # DOC foi removido para n?o confundir c?digos como DOC=100000.
    elif re.search(
        r"\bTRANSFERENCIA\b|\bTED\b|\bTEF\b",
        t_norm
    ):
        forma_pagamento = "Transfer\u00eancia"

    # Prioridade 6: Boleto (mant?m o comportamento anterior)
    elif re.search(
        r"\bBOLETO\b|LINHA\s+DIGITAVEL|CODIGO\s+DE\s+BARRAS",
        t_norm
    ):
        forma_pagamento = "Pix"

    logger.debug("EXTRATOR | forma_pagamento detectada: '%s'", forma_pagamento)

    # â”€â”€ Parcelas: identifica TOTAL (nÃ£o parcela atual) â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€â”€
    # Ordem de tentativa (mais especÃ­fico â†’ menos especÃ­fico):
    #   1. "Parcela N de M" ou "Parcela N/M" â†’ total = M
    #   2. "parc. N/M"  ou  "PARC N/M"      â†’ total = M
    #   3. "N/M" precedido por "PARCELA"     â†’ total = M
    #   4. "em Nx" / "Nx"                   â†’ total = N
    #   5. "N parcelas"                     â†’ total = N

    _parc_total: int | None = None

    # PadrÃ£o 1 â€” "Parcela 1 de 3" / "Parcela 1/3" / "1Âª parcela de 3"
    m = re.search(
        r"PARCELA\s+\d{1,2}\s+DE\s+(\d{1,2})"
        r"|PARCELA\s+\d{1,2}/(\d{1,2})"
        r"|\d{1,2}[ÂªÂº]\.?\s*PARCELA\s+DE\s+(\d{1,2})",
        t_upper
    )
    if m:
        g = next((x for x in m.groups() if x), None)
        if g:
            _parc_total = int(g)

    # PadrÃ£o 2 â€” "PARC. 01/03" / "PARC 1/3"
    if _parc_total is None:
        m = re.search(r"PARC\.?\s+\d{1,2}/(\d{1,2})", t_upper)
        if m:
            _parc_total = int(m.group(1))

    # PadrÃ£o 3 â€” "01/03" precedido de PARCELA na mesma linha
    if _parc_total is None:
        for linha in linhas:
            lu = linha.upper()
            if "PARCELA" in lu or "PARC" in lu:
                m = re.search(r"\b(\d{1,2})/(\d{1,2})\b", lu)
                if m:
                    num_atual, num_total = int(m.group(1)), int(m.group(2))
                    if num_total > num_atual and 2 <= num_total <= 48:
                        _parc_total = num_total
                        break

    # PadrÃ£o 4 â€” "em 3x", "3x", "3 x", "(3x)"
    if _parc_total is None:
        m = re.search(r"(?:EM\s+)?(\d{1,2})\s*[xX]\b", t_upper)
        if m:
            n = int(m.group(1))
            if 2 <= n <= 48:
                _parc_total = n

    # PadrÃ£o 5 â€” "em 3 parcelas" / "3 parcelas"
    if _parc_total is None:
        m = re.search(r"(?:EM\s+)?(\d{1,2})\s+PARCELA", t_upper)
        if m:
            n = int(m.group(1))
            if 2 <= n <= 48:
                _parc_total = n

    if _parc_total and 2 <= _parc_total <= 48:
        parcelas = _parc_total
        logger.debug("EXTRATOR | parcelas detectadas: %d", parcelas)

        # Se hÃ¡ parcelas e ainda nÃ£o identificamos a forma, verifica cartÃ£o
        if not forma_pagamento and "CART" in t_upper:
            forma_pagamento = "CartÃ£o de CrÃ©dito"
        elif forma_pagamento == "" and parcelas > 1:
            forma_pagamento = "CartÃ£o de CrÃ©dito"  # parcelamento â†’ crÃ©dito por padrÃ£o

    # ========================================================
    # 4. IDENTIFICAR VALOR TOTAL
    # ========================================================

    valor_total = 0.0

    # Procura por:
    # VALOR DO DOCUMENTO
    # VALOR TOTAL
    # TOTAL DA NOTA
    # VALOR COBRADO

    padroes_total = [

        r"(?:VALOR\s+DO\s+DOCUMENTO|VALOR\s+COBRADO|TOTAL\s+DA\s+NOTA|VALOR\s+TOTAL|TOTAL\s+A\s+PAGAR)[\s:=]*R?\$?\s*([\d\.,]+)",

        r"(?:VALOR|TOTAL)[\s:=]+R?\$?\s*([\d\.,]+)",

        r"R\$\s*([\d\.,]+)"
    ]

    for padrao in padroes_total:

        matches = re.findall(
            padrao,
            texto_completo,
            re.IGNORECASE
        )

        candidatos = []

        for match in matches:

            v_str = match.strip().replace(" ", "")

            # Trata formato brasileiro 1.234,56 ou 1234.56

            if "," in v_str and "." in v_str:

                if v_str.rfind(",") > v_str.rfind("."):
                    v_str = v_str.replace(".", "").replace(",", ".")

                else:
                    v_str = v_str.replace(",", "")

            elif "," in v_str:
                v_str = v_str.replace(",", ".")

            try:
                val = float(v_str)

                if val > 0:
                    candidatos.append(val)

            except ValueError:
                continue

        if candidatos:

            # Em boletos e DANFEs, o valor do documento
            # geralmente Ã© o maior ou o primeiro explÃ­cito

            valor_total = candidatos[0]
            break

    # ========================================================
    # 5. IDENTIFICAR ITENS
    # ========================================================

    itens = []

    # Tenta localizar linhas de produtos em DANFE ou lista simples

    palavras_bloqueadas = [

        "VALOR",
        "TOTAL",
        "DOCUMENTO",
        "MULTA",
        "JUROS",
        "DESCONTO",
        "PAGADOR",
        "BENEFICIARIO",
        "BENEFICIÃRIO",
        "AGÃŠNCIA",
        "AGENCIA",
        "CÃ“DIGO",
        "CODIGO",
        "CEDENTE",
        "SACADO",
        "COMPENSAÃ‡ÃƒO",
        "COMPENSACAO",
        "CNPJ",
        "CPF",
        "ENDEREÃ‡O",
        "ENDERECO",
        "REAL"
    ]

    for l in linhas:

        # Linhas contendo descriÃ§Ã£o e valor com R$

        m_item = re.search(
            r"^(.*?)\s+R\$\s*([\d\.,]+)$",
            l
        )

        if not m_item and " R$ " in l:

            partes = l.split(" R$ ")

            if len(partes) == 2:

                m_item = re.search(
                    r"^(.*?)\s+([\d\.,]+)$",
                    f"{partes[0]} {partes[1]}"
                )

        if m_item:

            desc = m_item.group(1).strip()

            # Ignora se for linha de boleto bancÃ¡rio ou cabeÃ§alho

            if any(
                kw in desc.upper()
                for kw in palavras_bloqueadas
            ):
                continue

            v_txt = m_item.group(2).strip()

            if "," in v_txt and "." in v_txt:
                v_txt = v_txt.replace(".", "").replace(",", ".")

            elif "," in v_txt:
                v_txt = v_txt.replace(",", ".")

            try:
                v_float = float(v_txt)

                # Valores de itens nÃ£o podem ser gigantescos
                # nem exceder o total

                if (
                    0.01 <= v_float <= (
                        valor_total if valor_total > 0 else 999999
                    )
                    and len(desc) > 2
                ):

                    itens.append({
                        'descricao': desc,
                        'valor': v_float,
                        'categoria_sugerida': ''
                    })

            except ValueError:
                pass

    if not itens and valor_total > 0:

        desc_padrao = fornecedor or "Despesa de Comprovante"

        itens.append({
            'descricao': desc_padrao,
            'valor': valor_total,
            'categoria_sugerida': ''
        })

    # ========================================================
    # RETORNO PADRONIZADO
    # ========================================================

    return {
        'fornecedor': fornecedor,
        'data': data_formatada,
        'forma_pagamento': forma_pagamento,
        'parcelas': parcelas,
        'valor_total': valor_total,
        'itens': itens
    }
